Run this automated Python script locally to download the MP4 video, isolate the soundtrack, and export page frames in a single run:

import os
import subprocess

YOUTUBE_URL = "https://www.youtube.com/watch?v=..."
FOLDER_NAME = "my-story"
TARGET_DIR = os.path.join("books", FOLDER_NAME)
PAGES_DIR = os.path.join(TARGET_DIR, "pages")

os.makedirs(PAGES_DIR, exist_ok=True)

# 1. Download Master MP4 Video
print("[1/3] Downloading MP4 video...")
video_path = os.path.join(TARGET_DIR, "story_video.mp4")
subprocess.run([
    "yt-dlp", "-f", "bv*[ext=mp4]+ba*[ext=m4a]/b[ext=mp4]/best",
    "--merge-output-format", "mp4",
    "-o", video_path,
    YOUTUBE_URL
], check=True)

# 2. Extract Soundtrack MP3
print("[2/3] Extracting soundtrack.mp3...")
subprocess.run([
    "yt-dlp", "-x", "--audio-format", "mp3",
    "-o", os.path.join(TARGET_DIR, "soundtrack.mp3"),
    YOUTUBE_URL
], check=True)

# 3. Extract High-Res Frame Slides
print("[3/3] Extracting page frames...")
subprocess.run([
    "ffmpeg", "-i", video_path,
    "-vf", "fps=1/4", "-q:v", "2",
    os.path.join(PAGES_DIR, "page-%03d.webp")
], check=True)

print("\nPreparation complete! MP4, MP3, and frames saved to:", TARGET_DIR)
